# Foundation Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the configuration service, the AI provider abstraction, the test doubles every later plan depends on, and the database schema.

**Architecture:** One `config.yaml` validated by a pydantic model is the single source of truth for every tunable; secrets come only from the environment. All AI calls go through two protocols — `LLMProvider` and `EmbeddingProvider` — so backends are swapped by config, and tests swap in deterministic fakes. Persistence is SQLite with an FTS5 virtual table for the keyword half of hybrid retrieval.

**Tech Stack:** Python 3.12, `uv`, pydantic v2, PyYAML, `anthropic`, `openai`, `sentence-transformers`, SQLAlchemy Core, `pytest`

## Global Constraints

- Python 3.12; dependencies via `uv`.
- Default LLM model is `claude-opus-5` for both classification and generation.
- Default embedding model is `all-MiniLM-L6-v2`, 384 dimensions, unit-normalized.
- Default confidence threshold 0.70; default relevance floor 0.35.
- Secrets (`ANTHROPIC_API_KEY`, `OPENAI_API_KEY`, `CREDENTIAL_ENCRYPTION_KEY`, `ADMIN_PASSWORD`) are read from the environment only and never written to `config.yaml`.
- Provider failures surface as `ProviderError` with category `timeout` | `rate_limit` | `auth` | `backend` — callers never see a backend-specific exception.
- L1 tests run with no network access.

from __future__ import annotations

from typing import Literal
import yaml
from pydantic import BaseModel, ConfigDict, Field, model_validator

class _Strict(BaseModel):
    model_config = ConfigDict(extra="forbid")

class LLMConfig(_Strict):
    provider: Literal["anthropic", "openai", "local"] = "anthropic"
    model_classify: str = "claude-opus-5"
    model_generate: str = "claude-opus-5"
    timeout_seconds: float = Field(default=20.0, gt=0)
    max_retries: int = Field(default=2, ge=0, le=10)

class EmbeddingConfig(_Strict):
    provider: Literal["local", "openai"] = "local"
    model: str = "all-MiniLM-L6-v2"
    dimension: int = Field(default=384, gt=0)
    batch_size: int = Field(default=64, gt=0)

class RAGConfig(_Strict):
    chunk_chars: int = Field(default=800, gt=0)
    chunk_overlap_chars: int = Field(default=100, ge=0)
    vector_top_n: int = Field(default=20, ge=0)
    keyword_top_n: int = Field(default=20, ge=0)
    rrf_k: int = Field(default=60, gt=0)
    final_top_k: int = Field(default=5, gt=0)
    max_context_chars: int = Field(default=6000, gt=0)
    relevance_floor: float = Field(default=0.35, ge=0.0, le=1.0)

    @model_validator(mode="after")
    def _overlap_smaller_than_chunk(self):
        if self.chunk_overlap_chars >= self.chunk_chars:
            raise ValueError("chunk_overlap_chars must be smaller than chunk_chars")
        return self

class OrchestratorConfig(_Strict):
    confidence_threshold: float = Field(default=0.70, ge=0.0, le=1.0)
    fallback_space: str = "general"

class IntentSpace(_Strict):
    slug: str = Field(pattern=r"^[a-z0-9]+(-[a-z0-9]+)*$")
    name: str
    description: str
    keywords: list[str] = Field(default_factory=list)

class ChannelConfig(_Strict):
    enabled: bool = False
    mode: Literal["polling", "webhook"] = "polling"
    max_message_chars: int = Field(default=4096, gt=0)

class ChannelsConfig(_Strict):
    telegram: ChannelConfig = ChannelConfig(enabled=True, mode="polling", max_message_chars=4096)
    teams: ChannelConfig = ChannelConfig(enabled=False, max_message_chars=28000)

class IngestionConfig(_Strict):
    max_upload_mb: int = Field(default=25, gt=0)
    allowed_extensions: list[str] = Field(default_factory=lambda: [".pdf", ".docx", ".xlsx"])

class StorageConfig(_Strict):
    sqlite_path: str = "./data/intelliknow.db"
    faiss_dir: str = "./data/faiss"
    upload_dir: str = "./data/uploads"

class AppConfig(_Strict):
    llm: LLMConfig = LLMConfig()
    embedding: EmbeddingConfig = EmbeddingConfig()
    rag: RAGConfig = RAGConfig()
    orchestrator: OrchestratorConfig = OrchestratorConfig()
    intent_spaces: list[IntentSpace]
    channels: ChannelsConfig = ChannelsConfig()
    ingestion: IngestionConfig = IngestionConfig()
    storage: StorageConfig = StorageConfig()
    public_base_url: str | None = None

    @model_validator(mode="after")
    def _fallback_space_exists(self):
        slugs = {s.slug for s in self.intent_spaces}
        if self.orchestrator.fallback_space not in slugs:
            raise ValueError(
                f"fallback_space '{self.orchestrator.fallback_space}' is not in intent_spaces")
        if len(slugs) != len(self.intent_spaces):
            raise ValueError("intent_space slugs must be unique")
        return self

    return AppConfig.model_validate(yaml.safe_load(Path(path).read_text()))

# app/config_service.py
from __future__ import annotations
import copy, os, tempfile
from pathlib import Path
import yaml
from pydantic import ValidationError
from app.config import AppConfig, load_config

def _deep_merge(base: dict, patch: dict) -> dict:
    out = copy.deepcopy(base)
    for k, v in patch.items():
        if isinstance(v, dict) and isinstance(out.get(k), dict):
            out[k] = _deep_merge(out[k], v)
        else:
            out[k] = v
    return out

class ConfigService:
    def __init__(self, path: Path, config: AppConfig) -> None:
        self._path = Path(path)
        self._config = config

    @classmethod
    def load(cls, path: Path) -> "ConfigService":
        return cls(path, load_config(path))

    @property
    def current(self) -> AppConfig:
        return self._config

    def reload(self) -> AppConfig:
        self._config = load_config(self._path)
        return self._config

    def update(self, patch: dict) -> AppConfig:
        raw = yaml.safe_load(self._path.read_text())
        merged = _deep_merge(raw, patch)
        try:
            validated = AppConfig.model_validate(merged)
        except ValidationError as exc:
            raise ValueError(str(exc)) from exc   # disk and memory untouched
        self._path.with_suffix(".yaml.bak").write_text(self._path.read_text())
        self._atomic_write(merged)
        self._config = validated
        return validated

    def _atomic_write(self, data: dict) -> None:
        fd, tmp = tempfile.mkstemp(dir=self._path.parent, suffix=".tmp")
        try:
            with os.fdopen(fd, "w") as fh:
                yaml.safe_dump(data, fh, sort_keys=False)
            os.replace(tmp, self._path)
        except BaseException:
            Path(tmp).unlink(missing_ok=True)
            raise
```

# app/providers/base.py
from __future__ import annotations
from dataclasses import dataclass
from typing import Literal, Protocol, runtime_checkable

Category = Literal["timeout", "rate_limit", "auth", "backend"]

@dataclass(frozen=True)
class LLMResult:
    text: str
    parsed: dict | None
    model: str
    input_tokens: int
    output_tokens: int

class ProviderError(Exception):
    def __init__(self, category: Category, message: str) -> None:
        super().__init__(f"[{category}] {message}")
        self.category: Category = category

    @classmethod
    def timeout(cls, m: str) -> "ProviderError": return cls("timeout", m)
    @classmethod
    def rate_limit(cls, m: str) -> "ProviderError": return cls("rate_limit", m)
    @classmethod
    def auth(cls, m: str) -> "ProviderError": return cls("auth", m)
    @classmethod
    def backend(cls, m: str) -> "ProviderError": return cls("backend", m)

@runtime_checkable
class LLMProvider(Protocol):
    def complete(self, *, system: str, user: str, schema: dict | None = None,
                 max_tokens: int = 1024) -> LLMResult: ...

@runtime_checkable
class EmbeddingProvider(Protocol):
    def embed(self, texts: list[str]) -> list[list[float]]: ...
    @property
    def dimension(self) -> int: ...

def normalize(vectors: list[list[float]]) -> list[list[float]]:
    out = []
    for v in vectors:
        norm = sum(x * x for x in v) ** 0.5
        out.append(list(v) if norm == 0 else [x / norm for x in v])
    return out
```

# tests/doubles.py
from __future__ import annotations
import hashlib
from collections import deque
from app.providers.base import LLMResult, ProviderError, normalize

class FakeLLMProvider:
    """Scripted LLM. Queue responses in order; every call is recorded for prompt assertions."""

    def __init__(self) -> None:
        self._queue: deque = deque()
        self.calls: list[dict] = []

    def expect_text(self, text: str) -> "FakeLLMProvider":
        self._queue.append(("text", text)); return self

    def expect_schema(self, parsed: dict) -> "FakeLLMProvider":
        self._queue.append(("parsed", parsed)); return self

    def fail_next(self, error: ProviderError) -> "FakeLLMProvider":
        self._queue.append(("error", error)); return self

    def complete(self, *, system: str, user: str, schema: dict | None = None,
                 max_tokens: int = 1024) -> LLMResult:
        self.calls.append({"system": system, "user": user, "schema": schema,
                           "max_tokens": max_tokens})
        assert self._queue, "no queued response on FakeLLMProvider"
        kind, payload = self._queue.popleft()
        if kind == "error":
            raise payload
        if kind == "parsed":
            return LLMResult(text="", parsed=payload, model="fake",
                             input_tokens=1, output_tokens=1)
        return LLMResult(text=payload, parsed=None, model="fake",
                         input_tokens=1, output_tokens=1)


class FakeEmbeddingProvider:
    """Deterministic hash-derived vectors. Same text always yields the same vector."""

    def __init__(self, dimension: int = 8) -> None:
        self._dimension = dimension
        self._pinned: dict[str, list[float]] = {}
        self.calls: list[list[str]] = []

    @property
    def dimension(self) -> int:
        return self._dimension

    def set_vector(self, text: str, vector: list[float]) -> None:
        assert len(vector) == self._dimension
        self._pinned[text] = vector

    def embed(self, texts: list[str]) -> list[list[float]]:
        self.calls.append(list(texts))
        raw = []
        for t in texts:
            if t in self._pinned:
                raw.append(self._pinned[t]); continue
            digest = hashlib.sha256(t.encode()).digest()
            raw.append([(digest[i % len(digest)] - 128) / 128.0 for i in range(self._dimension)])
        pinned_idx = {i for i, t in enumerate(texts) if t in self._pinned}
        normed = normalize(raw)
        return [raw[i] if i in pinned_idx else normed[i] for i in range(len(raw))]
```

# app/providers/retry.py
from __future__ import annotations
import time
from typing import Callable, TypeVar
from app.providers.base import ProviderError

T = TypeVar("T")
RETRYABLE = {"timeout", "rate_limit"}

def with_retries(fn: Callable[[], T], *, max_retries: int,
                 sleep: Callable[[float], None] = time.sleep) -> T:
    last: ProviderError | None = None
    for attempt in range(max_retries + 1):
        try:
            return fn()
        except ProviderError as exc:
            if exc.category not in RETRYABLE:
                raise
            last = exc
            if attempt < max_retries:
                sleep(0.5 * (2 ** attempt))
    assert last is not None
    raise last
```

# app/providers/anthropic_llm.py
from __future__ import annotations
import json
import anthropic
from app.providers.base import LLMResult, ProviderError
from app.providers.retry import with_retries

class AnthropicLLM:
    def __init__(self, model: str, api_key: str, timeout_seconds: float,
                 max_retries: int, client=None) -> None:
        self.model = model
        self._max_retries = max_retries
        self._client = client or anthropic.Anthropic(api_key=api_key, timeout=timeout_seconds,
                                                     max_retries=0)

    def complete(self, *, system: str, user: str, schema: dict | None = None,
                 max_tokens: int = 1024) -> LLMResult:
        def call() -> LLMResult:
            kwargs = {
                "model": self.model, "max_tokens": max_tokens, "system": system,
                "messages": [{"role": "user", "content": user}],
                "output_config": {"effort": "low"},
            }
            if schema is not None:
                kwargs["output_config"] = {
                    "effort": "low",
                    "format": {"type": "json_schema", "schema": schema},
                }
            try:
                resp = self._client.messages.create(**kwargs)
            except anthropic.AuthenticationError as e:
                raise ProviderError.auth(str(e)) from e
            except anthropic.RateLimitError as e:
                raise ProviderError.rate_limit(str(e)) from e
            except anthropic.APITimeoutError as e:
                raise ProviderError.timeout(str(e)) from e
            except anthropic.APIError as e:
                raise ProviderError.backend(str(e)) from e

            text = "".join(b.text for b in resp.content if getattr(b, "type", "") == "text")
            parsed = None
            if schema is not None:
                try:
                    parsed = json.loads(text)
                except json.JSONDecodeError as e:
                    raise ProviderError.backend(
                        f"response did not validate against requested schema: {e}") from e
            return LLMResult(text=text, parsed=parsed, model=resp.model,
                             input_tokens=resp.usage.input_tokens,
                             output_tokens=resp.usage.output_tokens)

        return with_retries(call, max_retries=self._max_retries)

from __future__ import annotations
from app.providers.base import normalize

class SentenceTransformerEmbedding:
    def __init__(self, model_name: str, batch_size: int) -> None:
        from sentence_transformers import SentenceTransformer
        self._model = SentenceTransformer(model_name)
        self._batch_size = batch_size
        self._dimension = int(self._model.get_sentence_embedding_dimension())

    @property
    def dimension(self) -> int:
        return self._dimension

    def embed(self, texts: list[str]) -> list[list[float]]:
        out: list[list[float]] = []
        for i in range(0, len(texts), self._batch_size):
            batch = texts[i:i + self._batch_size]
            out.extend(v.tolist() for v in self._model.encode(batch))
        return normalize(out)

SUPPORTED_EMBEDDING = ("local", "openai")

def _require(env: Mapping[str, str], name: str) -> str:
    value = env.get(name)
    if not value:
        raise RuntimeError(f"{name} is not set; required by the configured provider")
    return value

def build_llm_provider(cfg: AppConfig, *, role: Literal["classify", "generate"] = "generate",
                       env: Mapping[str, str] | None = None) -> LLMProvider:
    env = os.environ if env is None else env
    model = cfg.llm.model_classify if role == "classify" else cfg.llm.model_generate

        from app.providers.anthropic_llm import AnthropicLLM
        return AnthropicLLM(model, _require(env, "ANTHROPIC_API_KEY"),
                            cfg.llm.timeout_seconds, cfg.llm.max_retries)
    if cfg.llm.provider == "openai":
        from app.providers.openai_llm import OpenAILLM
        return OpenAILLM(model, _require(env, "OPENAI_API_KEY"),
                         cfg.llm.timeout_seconds, cfg.llm.max_retries)
    if cfg.llm.provider == "local":
        from app.providers.local_llm import LocalLLM
        return LocalLLM(model, cfg.llm.timeout_seconds, cfg.llm.max_retries)
    raise RuntimeError(
        f"unknown llm.provider '{cfg.llm.provider}'; supported: {', '.join(SUPPORTED_LLM)}")

def create_engine_for(path: Path):
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    engine = create_engine(f"sqlite:///{path}", future=True)

    @event.listens_for(engine, "connect")
    def _pragmas(dbapi_conn, _record):

        cur.execute("PRAGMA journal_mode=WAL")
        cur.execute("PRAGMA foreign_keys=ON")   # off by default in SQLite; cascade needs it

    return engine

    metadata.create_all(engine)
